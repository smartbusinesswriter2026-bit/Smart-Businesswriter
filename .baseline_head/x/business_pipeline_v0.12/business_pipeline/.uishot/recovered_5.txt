   215→MAX_ITEMS = 20
   216→MAX_PER_BLOCK = 5
   217→MAX_TEXT = 400
   218→
   219→
   220→def supported(template):
   221→    return template in SCHEMAS
   222→
   223→
   224→def _squash(value, limit):
   225→    text = re.sub(r"<[^>]+>", " ", str(value or ""))
   226→    text = re.sub(r"\s+", " ", text).strip()
   227→    return text[:limit]
   228→
   229→
   230→def _citation(source):
   231→    authors = _squash(source.get("authors"), 120)
   232→    year = source.get("year")
   233→    venue = _squash(source.get("venue"), 100)
   234→    link = source.get("doi") or source.get("url") or ""
   235→    parts = []
   236→    for part in (authors, str(year) if year else "", venue, link):
   237→        if part and part not in parts:
   238→            parts.append(part)
   239→    return _squash("; ".join(parts), 400)
   240→
   241→
   242→def _evidence(source):
   243→    title = _squash(source.get("title"), 220)
   244→    abstract = _squash(source.get("abstract") or source.get("description"), 800)
   245→    sentence = re.split(r"(?<=[.!?])\s+", abstract)[0] if abstract else ""
   246→    text = title + (" — " + sentence if sentence and sentence != title else "")
   247→    return _squash(text, MAX_TEXT)
   248→
   249→
   250→def _keys(source):
   251→    return set(re.findall(r"[a-z][a-z0-9'-]+", (
   252→        _squash(source.get("title"), 400) + " " + _squash(source.get("abstract") or source.get("description"), 1600)
   253→    ).lower()))
   254→
   255→
   256→def _best_block(schema, source):
   257→    words = _keys(source)
   258→    best, best_score = None, 0
   259→    for key, _label, cues in schema["blocks"]:
   260→        score = sum(1 for cue in cues.split() if cue in words)
   261→        if score > best_score:
   262→            best, best_score = key, score
   263→    return best, best_score
   264→
   265→
   266→def draft(template, sources, max_items=MAX_ITEMS):
   267→    """Return worksheet inputs plus a transparent placement report."""
   268→    schema = SCHEMAS.get(template)
   269→    if not schema:
   270→        raise ValueError("drafting is not supported for this worksheet yet")
   271→    if not isinstance(sources, list) or not sources:
   272→        raise ValueError("at least one retrieved source is required")
   273→
   274→    placed = {key: [] for key, _label, _cues in schema["blocks"]}
   275→    assigned, unmatched, seen = [], [], set()
   276→    for source in sources[:40]:
   277→        if not isinstance(source, dict) or not source.get("title"):
   278→            continue
   279→        key = source.get("doi") or _squash(source.get("title"), 120).lower()
   280→        if key in seen:
   281→            continue
   282→        seen.add(key)
   283→        block, score = _best_block(schema, source)
   284→        if not block or score == 0:
   285→            unmatched.append({"title": _squash(source.get("title"), 180), "reason": "no dimension matched"})
   286→            continue
   287→        if len(placed[block]) >= MAX_PER_BLOCK:
   288→            unmatched.append({"title": _squash(source.get("title"), 180), "reason": "block is full"})
   289→            continue
   290→        item = {"text": _evidence(source), "provenance": "sourced", "source": _citation(source)}
   291→        placed[block].append(item)
   292→        assigned.append({"block": block, "title": _squash(source.get("title"), 180),
   293→                         "citation": item["source"]})
   294→        if sum(len(items) for items in placed.values()) >= max_items:
   295→            break
   296→
   297→    if schema["kind"] == "swot":
   298→        inputs = {"evidence": [dict(item, dimension=key) for key, _label, _cues in schema["blocks"]
   299→                               for item in placed[key]], "options": []}
   300→    elif schema["kind"] == "forces":
   301→        inputs = {"evidence": [dict(item, force=key) for key, _label, _cues in schema["blocks"]
   302→                               for item in placed[key]], "ratings": {}}
   303→    else:
   304→        inputs = {"blocks": {key: placed[key] for key, _label, _cues in schema["blocks"]}, "links": []}
   305→
   306→    counts = {key: len(items) for key, items in placed.items()}
   307→    return {"template": template, "inputs": inputs, "assigned": assigned, "unmatched": unmatched,
   308→            "counts": counts, "labels": {key: label for key, label, _cues in schema["blocks"]},
   309→            "n_items": sum(counts.values()),
   310→            "note": ("Every drafted item is a source title plus a sentence from its abstract, with the "
   311→                     "citation recorded. Review it before running; no fact or number was invented.")}
